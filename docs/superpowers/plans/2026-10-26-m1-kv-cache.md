    device: str = "cuda:0",
    dtype: torch.dtype = torch.bfloat16,

```python
"""Naive per-layer KV cache for a single sequence (batch=1, M1).

Storage is concat-on-append: O(n) copy per token, O(n^2) per generation.
Intentionally naive -- correct and simple. M2's paged cache replaces the
storage behind the same update()/seq_len interface.
"""

from __future__ import annotations

import torch


class KVCache:
    def __init__(self, num_layers: int):
        self.k: list[torch.Tensor | None] = [None] * num_layers
        self.v: list[torch.Tensor | None] = [None] * num_layers

    @property
    def seq_len(self) -> int:
        """Cached length (read from layer 0; all layers must stay in lockstep)."""
        return 0 if self.k[0] is None else int(self.k[0].shape[2])

    def update(
        self, layer_idx: int, k_new: torch.Tensor, v_new: torch.Tensor
    ) -> tuple[torch.Tensor, torch.Tensor]:
        """Append this layer's new K/V; returns the full cached K/V.

        k_new/v_new: [1, kv_heads, new_len, head_dim]
        returns:     [1, kv_heads, past_len + new_len, head_dim]
        """
        if self.k[layer_idx] is None:
            self.k[layer_idx] = k_new
            self.v[layer_idx] = v_new
        else:
            self.k[layer_idx] = torch.cat((self.k[layer_idx], k_new), dim=2)
            self.v[layer_idx] = torch.cat((self.v[layer_idx], v_new), dim=2)
        return self.k[layer_idx], self.v[layer_idx]
```

### Task 4: cache 感知的 forward(掩码语义是本任务的核心知识点)

**Files:**
- Modify: `acorninfer/models/qwen3.py`(forward 穿 cache;attention 三分支掩码)
- Test: `tests/test_forward_cache.py`

掩码三分支(关键:SDPA 的 `is_causal=True` 是**左上对齐**,decode 的单 query 会被错掩到只看第一个 token,所以非空 cache 不能用它):
1. `past_len == 0`(prefill):`is_causal=(s > 1)`,flash 友好;
2. `s == 1`(decode):新 token 的 K/V 已入 cache,单 query 看得见全部 cache,**无掩码即正确**;
3. 其余(非空 cache + 多 token,如分块 prefill):显式右下对齐布尔掩码 `q_pos >= kv_pos`。

```python
    def forward(self, hidden, cos, sin, cache=None, layer_idx=0):
        b, s, _ = hidden.shape
        q = self.q_proj(hidden).view(b, s, self.num_heads, self.head_dim).transpose(1, 2)
        k = self.k_proj(hidden).view(b, s, self.num_kv_heads, self.head_dim).transpose(1, 2)
        v = self.v_proj(hidden).view(b, s, self.num_kv_heads, self.head_dim).transpose(1, 2)

        q = self.q_norm(q)
        k = self.k_norm(k)

        q, k = _apply_rope(q, k, cos, sin)

        if cache is not None:
            k, v = cache.update(layer_idx, k, v)  # k/v 变为全量 [1, kv, past+s, d]
        past_len = k.shape[2] - s

        k = _repeat_kv(k, self.num_heads // self.num_kv_heads)
        v = _repeat_kv(v, self.num_heads // self.num_kv_heads)

        if past_len == 0:
            # prefill:左上对齐的因果掩码 == 右下对齐(cache 空),flash 友好
            out = F.scaled_dot_product_attention(q, k, v, is_causal=(s > 1))
        elif s == 1:
            # decode:单 query,新 K/V 已在 cache 里,全可见 == 正确掩码
            out = F.scaled_dot_product_attention(q, k, v)
        else:
            # 非空 cache + 多 token:显式右下对齐掩码
            q_pos = torch.arange(past_len, past_len + s, device=q.device)
            kv_pos = torch.arange(past_len + s, device=q.device)
            mask = q_pos[:, None] >= kv_pos[None, :]  # True = 允许attend
            out = F.scaled_dot_product_attention(q, k, v, attn_mask=mask)

        out = out.transpose(1, 2).reshape(b, s, -1)
        return self.o_proj(out)
```

```python

        """input_ids: [batch, seq] -> logits: [batch, seq, vocab] (float32).

        cache=None: 全量前向(M0 oracle 路径)。传入 KVCache 时为增量路径:
        位置由 cache.seq_len 推导,prefill(cache 空)与 decode(cache 非空)共用。
        """

        past_len = 0 if cache is None else cache.seq_len
        positions = torch.arange(past_len, past_len + s, device=input_ids.device)
        dtype = self.embed_tokens.weight.dtype
        cos, sin = self.rope(positions, dtype, input_ids.device)
        hidden = self.embed_tokens(input_ids)
        for i, layer in enumerate(self.layers):
            hidden = layer(hidden, cos, sin, cache, i)
        hidden = self.norm(hidden)
        return self.lm_head(hidden).float()
```

```python
@torch.inference_mode()
def generate_cached(
    model: Qwen3ForCausalLM,
    prompt_ids: list[int],
    *,
    max_new_tokens: int = 64,
    eos_ids: frozenset[int] = EOS_IDS,
    sampler=None,
    cache: KVCache | None = None,
) -> tuple[list[int], list[float]]:
    """Prefill/decode generation with a KV cache: O(n) total instead of O(n^2).

    prefill  = one forward over the whole prompt (cache empty -> causal);
    decode   = one forward per new token (q_len == 1, attends to all cache).

    Returns (ids, step_logprobs): prompt included; step_logprobs holds the
    sampler's logprob for each chosen token (empty list when greedy).
    EOS semantics mirror HF: a generated EOS id is appended, then we stop.
    """
    if not prompt_ids:
        raise ValueError("prompt_ids must be non-empty")
    device = next(model.parameters()).device
    if cache is None:
        cache = KVCache(num_layers=len(model.layers))

    out = list(prompt_ids)
    logprobs: list[float] = []

    # prefill:整段 prompt 一次前向,取最后位置的分布
    last = model(torch.tensor([out], device=device, dtype=torch.long), cache)[0, -1]

    for _ in range(max_new_tokens):
        if sampler is None:
            next_id, lp = int(last.argmax()), None
        else:
            next_id, lp = sampler.sample(last)
        out.append(next_id)
        if lp is not None:
            logprobs.append(lp)
        if next_id in eos_ids:
            break
        # decode:单 token 增量前向
        last = model(torch.tensor([[next_id]], device=device, dtype=torch.long), cache)[
            0, -1
        ]
    return out, logprobs
```

```python
"""Sampling parameters and token selection.

Mirrors HF's warper pipeline order: temperature -> top-k -> top-p.
Numerical fidelity to HF is a hard requirement (M1 acceptance: per-step
logprob diff < 0.1); the unit suite pins our processing to HF's warpers.
"""

from __future__ import annotations

import torch
import torch.nn.functional as F


@dataclass(frozen=True)
class SamplingParams:
    temperature: float = 1.0
    top_p: float = 1.0
    top_k: int = 0  # 0 = disabled


class Sampler:
    def __init__(self, params: SamplingParams):
        if params.temperature <= 0:
            raise ValueError("temperature must be > 0")
        if not (0.0 < params.top_p <= 1.0):
            raise ValueError("top_p must be in (0, 1]")
        if params.top_k < 0:
            raise ValueError("top_k must be >= 0 (0 disables)")
        self.params = params

    def process(self, logits: torch.Tensor) -> torch.Tensor:
        """Apply temperature -> top-k -> top-p; returns processed [vocab] logits."""
        logits = logits.float() / self.params.temperature
        if self.params.top_k > 0:
            kth = torch.topk(logits, self.params.top_k).values[-1]
            logits[logits < kth] = float("-inf")
        if self.params.top_p < 1.0:
            sorted_logits, sorted_idx = torch.sort(logits, descending=False)
            cum = sorted_logits.softmax(dim=-1).cumsum(dim=-1)
            remove = cum <= (1.0 - self.params.top_p)
            remove[..., -1] = False  # 保底 1 个 token(HF min_tokens_to_keep=1)
            mask = torch.zeros_like(remove).scatter(0, sorted_idx, remove)
            logits[mask] = float("-inf")
        return logits

    def sample(self, logits: torch.Tensor) -> tuple[int, float]:
        """Process -> softmax -> multinomial; returns (token, its logprob)."""
        processed = self.process(logits)
        logprobs = F.log_softmax(processed, dim=-1)
        token = int(torch.multinomial(logprobs.exp(), num_samples=1))
        return token, float(logprobs[token])
```

方法学:测试 ② 用**教师强制**——把我们采样出的完整序列一次性喂给 HF 拿每步 logits,再用 HF 自己的 warper 处理,取我们所选 token 的 log_softmax 对比。这样两边在**相同上下文**下比较分布,绕开 RNG 流不可复现的问题。

```python
"""M1 acceptance: KV-cache path aligns with HF transformers.

- greedy 512: our first 256 generated ids are identical to HF's;
- sampling (temp 0.6 / top_p 0.95 / top_k 20): teacher-forced per-step
  logprob of OUR chosen token under HF's warpers, max abs diff < 0.1.

Run with CUDA_VISIBLE_DEVICES=1 (GPU 0 hosts an unrelated service);
sequential protocol: ours loads, runs, moves to CPU, then HF loads.
"""

import pytest
import torch
import transformers

from acorninfer.generate import generate_cached
from acorninfer.models.qwen3 import from_pretrained
from acorninfer.sampler import Sampler, SamplingParams
from acorninfer.tokenizer import Tokenizer

QWEN3_8B = "/home/ernest/model-weight/Qwen3-8B"
PROMPT = "The first three prime numbers are"
GREEDY_TOKENS = 512
ACCEPT_TOKENS = 256
SAMPLE_TOKENS = 48
PARAMS = SamplingParams(temperature=0.6, top_p=0.95, top_k=20)


def _free(model) -> None:
    model.to("cpu")
    torch.cuda.empty_cache()


def _load_hf():
    return transformers.AutoModelForCausalLM.from_pretrained(
        QWEN3_8B,
        dtype=torch.bfloat16,
        device_map={"": "cuda:0"},
        attn_implementation="sdpa",
    ).eval()


@pytest.mark.slow
def test_greedy_512_first_256_match_hf():
    tok = Tokenizer(f"{QWEN3_8B}/tokenizer.json")
    prompt_ids = tok.encode(PROMPT)

    ours = from_pretrained(QWEN3_8B)
    ids, _ = generate_cached(ours, prompt_ids, max_new_tokens=GREEDY_TOKENS)
    ours_new = ids[len(prompt_ids) :]
    _free(ours)

    hf = _load_hf()
    try:
        hf_ids = transformers.AutoTokenizer.from_pretrained(QWEN3_8B)(
            PROMPT, return_tensors="pt", add_special_tokens=False
        ).input_ids.to("cuda:0")
        hf_out = hf.generate(hf_ids, do_sample=False, max_new_tokens=ACCEPT_TOKENS)
        hf_new = hf_out[0][hf_ids.shape[1] :].tolist()
        assert len(ours_new) >= ACCEPT_TOKENS, f"ours stopped early at {len(ours_new)}"
        assert ours_new[:ACCEPT_TOKENS] == hf_new[:ACCEPT_TOKENS], (
            f"first divergence within 256 tokens:\nhf  = {hf_new[:ACCEPT_TOKENS]}\n"
            f"ours= {ours_new[:ACCEPT_TOKENS]}"
        )
    finally:
        _free(hf)


@pytest.mark.slow
def test_sampling_logprob_within_0_1_of_hf():
    tok = Tokenizer(f"{QWEN3_8B}/tokenizer.json")
    prompt_ids = tok.encode(PROMPT)

    ours = from_pretrained(QWEN3_8B)
    ids, step_lps = generate_cached(
        ours, prompt_ids, max_new_tokens=SAMPLE_TOKENS, sampler=Sampler(PARAMS)
    )
    assert len(step_lps) > 0
    # 注意:尾部 EOS 不剥掉 —— 教师强制的 last logprob 对应"选 EOS 那一步"的分布,
    # 剥掉会令最后一个 pos+1 越界;带 EOS 的序列照常前向即可。
    _free(ours)

    hf = _load_hf()
    try:
        full = torch.tensor([ids], device="cuda:0")
        with torch.inference_mode():
            hf_logits = hf(full).logits[0].float()  # [T, vocab]
        warpers = hf._get_logits_warper(
            transformers.GenerationConfig(
                temperature=PARAMS.temperature,
                top_p=PARAMS.top_p,
                top_k=PARAMS.top_k,
                do_sample=True,
            )
        )
        max_diff = 0.0
        for t, our_lp in enumerate(step_lps):
            pos = len(prompt_ids) - 1 + t  # 位置 pos 的分布预测 token pos+1
            scores = hf_logits[pos][None]
            for w in warpers:
                scores = w(full[:, : pos + 1], scores)
            hf_lp = float(torch.log_softmax(scores[0], -1)[ids[pos + 1]])
            max_diff = max(max_diff, abs(hf_lp - our_lp))
        assert max_diff < 0.1, f"per-step logprob max abs diff = {max_diff}"
    finally:
        _free(hf)
```
