```python
"""Model configuration."""

from __future__ import annotations

import json
from dataclasses import dataclass
from pathlib import Path


@dataclass(frozen=True)
class Qwen3Config:
    num_hidden_layers: int
    hidden_size: int
    num_attention_heads: int
    num_key_value_heads: int
    head_dim: int
    intermediate_size: int
    rms_norm_eps: float
    rope_theta: float
    vocab_size: int
    tie_word_embeddings: bool
    bos_token_id: int
    eos_token_id: list[int]

    @classmethod
    def from_json(cls, path: Path | str) -> Qwen3Config:
        raw = json.loads(Path(path).read_text())
        eos = raw["eos_token_id"]
        return cls(
            num_hidden_layers=raw["num_hidden_layers"],
            hidden_size=raw["hidden_size"],
            num_attention_heads=raw["num_attention_heads"],
            num_key_value_heads=raw["num_key_value_heads"],
            head_dim=raw["head_dim"],
            intermediate_size=raw["intermediate_size"],
            rms_norm_eps=raw["rms_norm_eps"],
            rope_theta=raw["rope_theta"],
            vocab_size=raw["vocab_size"],
            tie_word_embeddings=raw["tie_word_embeddings"],
            bos_token_id=raw["bos_token_id"],
            eos_token_id=eos if isinstance(eos, list) else [eos],
        )

    @classmethod
    def tiny(cls) -> Qwen3Config:
        """Small config for cheap unit tests."""
        return cls(
            num_hidden_layers=2,

            num_attention_heads=4,
            num_key_value_heads=2,
```

```python
"""Loading model weights from a safetensors sharded checkpoint."""

from __future__ import annotations

import json
from pathlib import Path

import torch
from safetensors.torch import load_file


def load_safetensors(
    model_dir: str | Path,
    device: str | torch.device = "cpu",
    dtype: torch.dtype = torch.bfloat16,
) -> dict[str, torch.Tensor]:
    """Load all shards named by model.safetensors.index.json into one dict."""
    model_dir = Path(model_dir)
    index = json.loads((model_dir / "model.safetensors.index.json").read_text())
    shard_names = sorted(set(index["weight_map"].values()))

    tensors: dict[str, torch.Tensor] = {}
    for shard in shard_names:
        loaded = load_file(model_dir / shard, device=str(device))
        tensors.update(loaded)
    return {name: t.to(dtype=dtype) for name, t in tensors.items()}
```

```python
"""Thin wrapper around the HF tokenizers library.

The engine's own tokenization path; `transformers` is never imported here.
"""

from __future__ import annotations

from tokenizers import Tokenizer as _HFTokenizer


class Tokenizer:
    def __init__(self, tokenizer_json: str):
        self._tk = _HFTokenizer.from_file(tokenizer_json)

    def encode(self, text: str) -> list[int]:
        """Text to token ids; no special tokens are added."""
        return self._tk.encode(text, add_special_tokens=False).ids

    def decode(self, ids: list[int]) -> str:
        """Token ids to text, skipping special tokens."""
        return self._tk.decode(ids, skip_special_tokens=True)
```

```python
"""Qwen3 dense model, pure-PyTorch reference implementation (no KV cache).

Numerics deliberately mirror HF transformers Qwen3:
- RMSNorm computes in float32 and casts back to the input dtype;
- RoPE cos/sin are computed in float32 and cast to the activation dtype
  before the elementwise multiply;
- QK-Norm normalizes each head over head_dim;
- SDPA with is_causal=True and the default softmax scale (head_dim ** -0.5).
"""

from __future__ import annotations

import torch
import torch.nn as nn
import torch.nn.functional as F

from suiren.config import Qwen3Config


class RMSNorm(nn.Module):
    def __init__(self, dim: int, eps: float):
        super().__init__()
        self.eps = eps
        self.weight = nn.Parameter(torch.ones(dim))

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        dtype = x.dtype
        x = x.float()
        x = x * torch.rsqrt(x.pow(2).mean(-1, keepdim=True) + self.eps)
        return (x * self.weight.float()).to(dtype)


def _rotate_half(x: torch.Tensor) -> torch.Tensor:
    x1, x2 = x.chunk(2, dim=-1)
    return torch.cat((-x2, x1), dim=-1)


class RotaryEmbedding(nn.Module):
    def __init__(self, head_dim: int, theta: float):
        super().__init__()
        inv_freq = 1.0 / (
            theta ** (torch.arange(0, head_dim, 2, dtype=torch.float32) / head_dim)
        )
        self.register_buffer("inv_freq", inv_freq, persistent=False)

    def forward(self, seq_len: int, dtype: torch.dtype, device: torch.device):
        pos = torch.arange(seq_len, device=device, dtype=torch.float32)
        # `.to(dtype=...)` on the module casts this buffer to bf16; force fp32 back.
        freqs = torch.outer(pos, self.inv_freq.float().to(device))
        emb = torch.cat((freqs, freqs), dim=-1)
        cos = emb.cos().to(dtype)
        sin = emb.sin().to(dtype)
        return cos, sin  # [seq_len, head_dim]


def _apply_rope(
    q: torch.Tensor, k: torch.Tensor, cos: torch.Tensor, sin: torch.Tensor
) -> tuple[torch.Tensor, torch.Tensor]:
    # q/k: [batch, heads, seq, head_dim]; cos/sin: [seq, head_dim]
    cos = cos[None, None, :, :]
    sin = sin[None, None, :, :]
    q = q * cos + _rotate_half(q) * sin
    k = k * cos + _rotate_half(k) * sin
    return q, k


def _repeat_kv(x: torch.Tensor, repeat: int) -> torch.Tensor:
    # [batch, kv_heads, seq, dim] -> [batch, kv_heads * repeat, seq, dim]
    if repeat == 1:
        return x
    b, h, s, d = x.shape
    return x[:, :, None, :, :].expand(b, h, repeat, s, d).reshape(b, h * repeat, s, d)


class Qwen3Attention(nn.Module):
    def __init__(self, cfg: Qwen3Config):
        super().__init__()
        self.num_heads = cfg.num_attention_heads
        self.num_kv_heads = cfg.num_key_value_heads
        self.head_dim = cfg.head_dim
        self.q_proj = nn.Linear(cfg.hidden_size, self.num_heads * self.head_dim, bias=False)
        self.k_proj = nn.Linear(cfg.hidden_size, self.num_kv_heads * self.head_dim, bias=False)
        self.v_proj = nn.Linear(cfg.hidden_size, self.num_kv_heads * self.head_dim, bias=False)
        self.o_proj = nn.Linear(self.num_heads * self.head_dim, cfg.hidden_size, bias=False)
        self.q_norm = RMSNorm(self.head_dim, cfg.rms_norm_eps)
        self.k_norm = RMSNorm(self.head_dim, cfg.rms_norm_eps)
        self.rope = RotaryEmbedding(cfg.head_dim, cfg.rope_theta)

    def forward(self, hidden: torch.Tensor) -> torch.Tensor:
        b, s, _ = hidden.shape
        q = self.q_proj(hidden).view(b, s, self.num_heads, self.head_dim).transpose(1, 2)
        k = self.k_proj(hidden).view(b, s, self.num_kv_heads, self.head_dim).transpose(1, 2)
        v = self.v_proj(hidden).view(b, s, self.num_kv_heads, self.head_dim).transpose(1, 2)

        q = self.q_norm(q)
        k = self.k_norm(k)

        cos, sin = self.rope(s, hidden.dtype, hidden.device)
        q, k = _apply_rope(q, k, cos, sin)

        k = _repeat_kv(k, self.num_heads // self.num_kv_heads)
        v = _repeat_kv(v, self.num_heads // self.num_kv_heads)

        out = F.scaled_dot_product_attention(q, k, v, is_causal=True)
        out = out.transpose(1, 2).reshape(b, s, -1)
        return self.o_proj(out)


class Qwen3MLP(nn.Module):
    def __init__(self, cfg: Qwen3Config):
        super().__init__()
        self.gate_proj = nn.Linear(cfg.hidden_size, cfg.intermediate_size, bias=False)
        self.up_proj = nn.Linear(cfg.hidden_size, cfg.intermediate_size, bias=False)
        self.down_proj = nn.Linear(cfg.intermediate_size, cfg.hidden_size, bias=False)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return self.down_proj(F.silu(self.gate_proj(x)) * self.up_proj(x))


class Qwen3Block(nn.Module):
    def __init__(self, cfg: Qwen3Config):
        super().__init__()
        self.input_layernorm = RMSNorm(cfg.hidden_size, cfg.rms_norm_eps)
        self.self_attn = Qwen3Attention(cfg)
        self.post_attention_layernorm = RMSNorm(cfg.hidden_size, cfg.rms_norm_eps)
        self.mlp = Qwen3MLP(cfg)

    def forward(self, hidden: torch.Tensor) -> torch.Tensor:
        hidden = hidden + self.self_attn(self.input_layernorm(hidden))
        hidden = hidden + self.mlp(self.post_attention_layernorm(hidden))
        return hidden


class Qwen3ForCausalLM(nn.Module):
    def __init__(self, cfg: Qwen3Config):
        super().__init__()
        self.cfg = cfg
        self.embed_tokens = nn.Embedding(cfg.vocab_size, cfg.hidden_size)
        self.layers = nn.ModuleList(Qwen3Block(cfg) for _ in range(cfg.num_hidden_layers))
        self.norm = RMSNorm(cfg.hidden_size, cfg.rms_norm_eps)
        self.lm_head = nn.Linear(cfg.hidden_size, cfg.vocab_size, bias=False)

    def forward(self, input_ids: torch.Tensor) -> torch.Tensor:
        """input_ids: [batch, seq] -> logits: [batch, seq, vocab] (float32)."""
        hidden = self.embed_tokens(input_ids)
        for layer in self.layers:
            hidden = layer(hidden)
        hidden = self.norm(hidden)
        return self.lm_head(hidden).float()


def map_hf_state_dict(hf_tensors: dict[str, torch.Tensor]) -> dict[str, torch.Tensor]:
    """HF checkpoint names -> our module names (strip the `model.` prefix)."""
    return {name.removeprefix("model."): t for name, t in hf_tensors.items()}
```

```python
"""Greedy generation without a KV cache (M0).

Each step re-runs the full sequence: O(n^2) total, correct and simple.
The KV-cache path replaces this in M1; this loop is the golden oracle.
"""

from __future__ import annotations

import torch

from suiren.models.qwen3 import Qwen3ForCausalLM
from suiren.tokenizer import Tokenizer

EOS_IDS = frozenset({151643, 151645})  # <|endoftext|>, <|im_end|>


@torch.inference_mode()
def generate_token_ids(
    model: Qwen3ForCausalLM,
    prompt_ids: list[int] | None = None,
    *,
    tokenizer: Tokenizer | None = None,
    prompt: str | None = None,
    max_new_tokens: int = 64,
) -> list[int]:
    """Generate greedily; returns prompt ids + generated ids (prompt included)."""
    if prompt_ids is None:
        assert tokenizer is not None and prompt is not None
        prompt_ids = tokenizer.encode(prompt)
    device = next(model.parameters()).device

    out = list(prompt_ids)
    for _ in range(max_new_tokens):
        input_ids = torch.tensor([out], device=device, dtype=torch.long)
        logits = model(input_ids)
        next_id = int(logits[0, -1].argmax())
        if next_id in EOS_IDS:
            break
        out.append(next_id)
    return out
```

```python
"""M0 acceptance: our greedy tokens match HF transformers token-for-token.

Spec: on fixed prompts, the first >= 50 generated tokens are identical.
"""

import pytest
import torch
import transformers

from suiren.config import Qwen3Config
from suiren.generate import generate_token_ids
from suiren.models.qwen3 import Qwen3ForCausalLM, map_hf_state_dict
from suiren.tokenizer import Tokenizer
from suiren.weights import load_safetensors

QWEN3_8B = "/home/ernest/model-weight/Qwen3-8B"
PROMPTS = [
    "The capital of France is",
    "The first three prime numbers are",
    "Water boils at a temperature of",
]
ACCEPT_TOKENS = 50


def load_suiren() -> Qwen3ForCausalLM:
    cfg = Qwen3Config.from_json(f"{QWEN3_8B}/config.json")
    model = Qwen3ForCausalLM(cfg).to(device="cuda:0", dtype=torch.bfloat16)
    mapped = map_hf_state_dict(load_safetensors(QWEN3_8B, device="cpu"))
    model.load_state_dict(mapped, strict=True, assign=True)
    return model.eval()


def load_hf() -> transformers.PreTrainedModel:
    return transformers.AutoModelForCausalLM.from_pretrained(
        QWEN3_8B,
        dtype=torch.bfloat16,
        device_map={"": "cuda:1"},
        attn_implementation="sdpa",
    ).eval()


@pytest.mark.slow
def test_golden_alignment():
    suiren_model = load_suiren()
    hf_model = load_hf()
    hf_tok = transformers.AutoTokenizer.from_pretrained(QWEN3_8B)
    suiren_tok = Tokenizer(f"{QWEN3_8B}/tokenizer.json")

    for prompt in PROMPTS:
        hf_ids = hf_tok(prompt, return_tensors="pt", add_special_tokens=False).input_ids.to(
            "cuda:1"
        )
        hf_out = hf_model.generate(
            hf_ids, do_sample=False, max_new_tokens=ACCEPT_TOKENS + 10
        )
        hf_new = hf_out[0][hf_ids.shape[1]:].tolist()

        ours_all = generate_token_ids(
            suiren_model, tokenizer=suiren_tok, prompt=prompt, max_new_tokens=ACCEPT_TOKENS + 10
        )
        ours_new = ours_all[len(suiren_tok.encode(prompt)):]

        matched = sum(a == b for a, b in zip(hf_new, ours_new))
        assert matched >= ACCEPT_TOKENS, (
            f"prompt={prompt!r} matched {matched}/{ACCEPT_TOKENS};\n"
            f"hf  = {hf_new[:60]}\nours= {ours_new[:60]}"
        )

    del suiren_model, hf_model  # 释放两张卡的 16GB
    torch.cuda.empty_cache()
```

```python
"""`python -m suiren` command-line entry."""

from __future__ import annotations

import argparse
import time

import torch

from suiren.config import Qwen3Config
from suiren.generate import generate_token_ids
from suiren.models.qwen3 import Qwen3ForCausalLM, map_hf_state_dict
from suiren.tokenizer import Tokenizer
from suiren.weights import load_safetensors

DEFAULT_MODEL = "/home/ernest/model-weight/Qwen3-8B"


def main() -> None:
    parser = argparse.ArgumentParser(prog="suiren")
    sub = parser.add_subparsers(dest="cmd", required=True)
    gen = sub.add_parser("generate", help="greedy generation (M0)")
    gen.add_argument("--prompt", required=True)
    gen.add_argument("--model-dir", default=DEFAULT_MODEL)
    gen.add_argument("--max-new-tokens", type=int, default=64)
    args = parser.parse_args()

    device = "cuda:0"
    t0 = time.perf_counter()
    cfg = Qwen3Config.from_json(f"{args.model_dir}/config.json")
    model = Qwen3ForCausalLM(cfg).to(device=device, dtype=torch.bfloat16)
    mapped = map_hf_state_dict(load_safetensors(args.model_dir, device="cpu"))
    model.load_state_dict(mapped, strict=True, assign=True)
    model.eval()
    print(f"model loaded in {time.perf_counter() - t0:.1f}s")

    tok = Tokenizer(f"{args.model_dir}/tokenizer.json")
    t0 = time.perf_counter()
    ids = generate_token_ids(
        model, tokenizer=tok, prompt=args.prompt, max_new_tokens=args.max_new_tokens
    )
    elapsed = time.perf_counter() - t0
    prompt_len = len(tok.encode(args.prompt))
    new_tokens = len(ids) - prompt_len
    print(f"\n{tok.decode(ids)}")
    print(f"\n--- {new_tokens} tokens in {elapsed:.1f}s ({new_tokens / elapsed:.2f} tok/s, 无 cache 全量重算)")
```
